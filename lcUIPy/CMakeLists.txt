CMAKE_MINIMUM_REQUIRED(VERSION 3.6...3.27)

message("***** LibreCAD Python scripts *****")

# Phase 5 PR-5.3 — path bootstrap.
#
# Mirrors lcUILua/CMakeLists.txt.  Generates `path.py` at build time in
# the runtime output directory (alongside `path.lua`).  C++ side loads
# it during LuaInterface::initLua so `import lcUIPy.operations` and
# friends work.  The install-tree variant writes the INSTALLED path
# (fixing the same "baked build-machine source path" bug present in
# lcUILua/CMakeLists.txt lines 16/28 today).
#
# Content structure (mirrors path.lua):
#   * Extends sys.path with the lcUIPy dir so `import lcUIPy.<mod>` works.
#   * `plugin_path` global for Python plugins (used by PluginManager in
#     PR-5.5).
#   * `python_path` global — the analog of lua_path — points at the
#     lcUIPy install/source root.
#   * Unittest-binary fallback: if the primary path doesn't exist,
#     walk up looking for `lcUIPy/` — the Python analog of Lua's
#     FolderFinder fallback (operationloader.cpp:22-83).
#
# NOTE the sub-plan called out the following install bugs in path.lua
# (lcUILua/CMakeLists.txt:16 and :28) — SAME bug pattern, both fixed
# here for path.py; a separate 1-line lcUILua/CMakeLists.txt patch
# below fixes the Lua sibling.

if(WITH_PYTHONSCRIPT)

# ---- Build-tree path.py ------------------------------------------------------
file(WRITE "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/path.py"
"# path.py — LibreCAD Python bootstrap.  Auto-generated by
# lcUIPy/CMakeLists.txt; do NOT edit by hand.
import os
import sys

# Primary source root (build-tree).
_primary_root = '${CMAKE_CURRENT_LIST_DIR}'

# Fallback for unittest binaries whose runtime dir isn't the librecad
# bin dir — walk up looking for a sibling lcUIPy/.  Mirrors what
# FolderFinder does for the Lua side in operationloader.cpp.
def _find_lcuipy():
    if os.path.isdir(_primary_root):
        return _primary_root
    here = os.path.dirname(os.path.abspath(__file__))
    for _ in range(6):
        candidate = os.path.join(here, 'lcUIPy')
        if os.path.isdir(candidate):
            return candidate
        here = os.path.dirname(here)
    return _primary_root  # last resort — import will fail loudly

python_path = _find_lcuipy()
# Insert the PARENT of lcUIPy so `import lcUIPy.operations` resolves.
_parent = os.path.dirname(python_path)
if _parent not in sys.path:
    sys.path.insert(0, _parent)

# Python plugin directory (used by PluginManager, PR-5.5).
plugin_path = '${PROJECT_SOURCE_DIR}/lcUIPy/plugins'
")

# ---- Install-tree path.py ---------------------------------------------------
# The install-tree write MUST bake the INSTALLED path, not the build
# machine's PROJECT_SOURCE_DIR — same bug class as lcUILua's fix below.
if(MSVC)
    install(CODE "file(WRITE \"${FINAL_INSTALL_DIR}/${CMAKE_INSTALL_BINDIR}/path.py\"
\"# path.py — auto-generated (install tree).\n\"
\"import os, sys\n\"
\"python_path = '${FINAL_INSTALL_DIR}/lcUIPy'\n\"
\"_parent = os.path.dirname(python_path)\n\"
\"if _parent not in sys.path:\n\"
\"    sys.path.insert(0, _parent)\n\"
\"plugin_path = '${FINAL_INSTALL_DIR}/lcUIPy/plugins'\n\"
    )")

    install(DIRECTORY "${CMAKE_SOURCE_DIR}/lcUIPy"
            DESTINATION "${FINAL_INSTALL_DIR}/"
            PATTERN "CMakeLists.txt" EXCLUDE
            PATTERN "__pycache__" EXCLUDE
    )
else()
    install(CODE "file(WRITE \"${FINAL_INSTALL_DIR}/${CMAKE_INSTALL_BINDIR}/path.py\"
\"# path.py — auto-generated (install tree).\n\"
\"import os, sys\n\"
\"python_path = '${FINAL_INSTALL_DIR}/usr/share/librecad/lcUIPy'\n\"
\"_parent = os.path.dirname(python_path)\n\"
\"if _parent not in sys.path:\n\"
\"    sys.path.insert(0, _parent)\n\"
\"plugin_path = '${FINAL_INSTALL_DIR}/usr/share/librecad/lcUIPy/plugins'\n\"
    )")

    install(DIRECTORY "${CMAKE_SOURCE_DIR}/lcUIPy"
            DESTINATION "${CMAKE_INSTALL_BINDIR}/../share/librecad"
            PATTERN "CMakeLists.txt" EXCLUDE
            PATTERN "__pycache__" EXCLUDE
    )
endif()

endif()  # WITH_PYTHONSCRIPT
