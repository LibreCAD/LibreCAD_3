foreach(i RANGE ${last})
    string(JSON id GET "${requirements}" fixtures ${i} id)
    if(id STREQUAL "abeezee")
        math(EXPR found "${found} + 1")
        foreach(field path sha256 notice notice_sha256)
            string(JSON font_${field} GET "${requirements}" fixtures ${i} ${field})

    endif()

    message(FATAL_ERROR "Expected one pinned ABeeZee fixture")

    if(IS_ABSOLUTE "${font_${field}}" OR "${font_${field}}" MATCHES "(^|/)\\.\\.(/|$)")

    endif()

function(expected variant name result)
    string(JSON count LENGTH "${contracts}" datasets text ${variant} assertions)
    math(EXPR last "${count} - 1")
    set(matches 0)
    foreach(i RANGE ${last})
        string(JSON key GET "${contracts}" datasets text ${variant} assertions ${i} observed)
        if(key STREQUAL name)
            math(EXPR matches "${matches} + 1")
            string(JSON type TYPE "${contracts}" datasets text ${variant} assertions ${i} expected value)
            if(type STREQUAL "ARRAY")
                string(JSON size LENGTH "${contracts}" datasets text ${variant} assertions ${i} expected value)
                if(NOT size EQUAL 1)
                    message(FATAL_ERROR "This bounded audit expects one glyph per fixture")
                endif()
                string(JSON type TYPE "${contracts}" datasets text ${variant} assertions ${i} expected value 0)
                string(JSON value GET "${contracts}" datasets text ${variant} assertions ${i} expected value 0)
            else()
                string(JSON value GET "${contracts}" datasets text ${variant} assertions ${i} expected value)
            endif()
            string(JSON unit GET "${contracts}" datasets text ${variant} assertions ${i} expected unit)
            if(name MATCHES "^raw-")
                set(wanted_unit font-unit)
            elseif(name STREQUAL "decoded-codepoints")
                set(wanted_unit codepoint)
            else()
                set(wanted_unit count)
            endif()
            if(NOT type STREQUAL "NUMBER" OR NOT unit STREQUAL wanted_unit)
                message(FATAL_ERROR "Wrong font oracle type/unit: ${variant}/${name}")
            endif()
        endif()
    endforeach()
    if(NOT matches EQUAL 1 OR NOT "${value}" MATCHES "^[0-9]+$")
        message(FATAL_ERROR "Missing/ambiguous integer font oracle: ${variant}/${name}")
    endif()
    set(${result} "${value}" PARENT_SCOPE)
endfunction()
string(JSON FAMILY GET "${contracts}" datasets text ordinary inputs font-family value)
string(JSON STYLE GET "${contracts}" datasets text ordinary inputs font-style value)
string(JSON UPEM GET "${contracts}" datasets text ordinary inputs units-per-em value)
string(JSON upem_type TYPE "${contracts}" datasets text ordinary inputs units-per-em value)
if(NOT upem_type STREQUAL "NUMBER" OR NOT "${UPEM}" MATCHES "^[0-9]+$" OR UPEM LESS 1 OR UPEM GREATER 65535)
    message(FATAL_ERROR "Invalid design-unit scale")
endif()
string(JSON ordinary_bytes LENGTH "${contracts}" datasets text ordinary inputs utf8-bytes value)
string(JSON ordinary_cp GET "${contracts}" datasets text ordinary inputs utf8-bytes value 0)
if(NOT ordinary_bytes EQUAL 1 OR ordinary_cp GREATER 127)
    message(FATAL_ERROR "Ordinary probe operand must be one ASCII codepoint, not a UTF-8 decoder")
endif()
set(PROBES "")
foreach(variant ordinary latin supplementary)
    if(variant STREQUAL "ordinary")
        set(cp ${ordinary_cp})
        expected(${variant} raw-advances advance)
    else()
        expected(${variant} decoded-codepoints cp)
        expected(${variant} raw-advance advance)
    endif()
    expected(${variant} glyph-ids glyph)
    string(APPEND PROBES "{\"${variant}\", ${cp}, ${glyph}, ${advance}},\n")
endforeach()
expected(ordinary raw-left-bearings LEFT_BEARING)

    get_target_property(provider Freetype::Freetype IMPORTED_LOCATION)

    message(FATAL_ERROR "Expected an identifiable prebuilt FreeType library")
endif()
