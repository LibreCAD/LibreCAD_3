cmake_minimum_required(VERSION 3.21)
project(LibreCADR0QtProvider LANGUAGES CXX)

# A binary-provider audit, never a production renderer or support floor.
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$")
    message(FATAL_ERROR "This R0 provider probe qualifies Linux and macOS only")
endif()
set(PROBE_QT_PREFIX "" CACHE PATH "Fresh private Qt installation prefix")
set(PROBE_QT_VERSION "" CACHE STRING "Exact Qt version under test")
if(NOT IS_DIRECTORY "${PROBE_QT_PREFIX}" OR NOT IS_ABSOLUTE "${PROBE_QT_PREFIX}")
    message(FATAL_ERROR "Provider prefix must be an existing absolute directory")
endif()
if(NOT PROBE_QT_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "An exact PROBE_QT_VERSION is required")
endif()
file(REAL_PATH "${PROBE_QT_PREFIX}" probe_prefix)
function(require_provider_path label path)
    if(NOT EXISTS "${path}" OR NOT IS_ABSOLUTE "${path}")
        message(FATAL_ERROR "Missing provider input: ${label}: ${path}")
    endif()
    file(REAL_PATH "${path}" real_path)
    cmake_path(IS_PREFIX probe_prefix "${real_path}" NORMALIZE inside)
    if(NOT inside)
        message(FATAL_ERROR "Provider prefix mismatch: ${label}: ${real_path}")
    endif()
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/provider-imports.txt" "${label}=${real_path}\n")
endfunction()

set(QT_NO_CREATE_VERSIONLESS_TARGETS ON)
set(modules Core Gui Widgets Svg SvgWidgets UiTools Test OpenGL OpenGLWidgets ShaderTools)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/provider-imports.txt" "requested_qt_version=${PROBE_QT_VERSION}\n")
if(DEFINED Qt6_DIR)
    require_provider_path(Qt6_DIR "${Qt6_DIR}")
endif()
find_package(Qt6 ${PROBE_QT_VERSION} EXACT REQUIRED COMPONENTS ${modules})
if(NOT TARGET Qt6::GuiPrivate)
    find_package(Qt6 ${PROBE_QT_VERSION} EXACT REQUIRED COMPONENTS GuiPrivate)
    list(APPEND modules GuiPrivate)
endif()
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/provider-imports.txt" "qt_version=${Qt6_VERSION}\n")
require_provider_path(Qt6_DIR "${Qt6_DIR}")
foreach(module IN LISTS modules)
    require_provider_path("Qt6${module}_DIR" "${Qt6${module}_DIR}")
endforeach()
foreach(target Core Gui Widgets Svg SvgWidgets UiTools Test OpenGL OpenGLWidgets ShaderTools qsb moc rcc uic)
    if(NOT TARGET Qt6::${target})
        message(FATAL_ERROR "Missing provider target Qt6::${target}")
    endif()
    get_target_property(configurations Qt6::${target} IMPORTED_CONFIGURATIONS)
    set(properties IMPORTED_LOCATION)
    foreach(config IN LISTS configurations)
        string(TOUPPER "${config}" upper)
        list(APPEND properties "IMPORTED_LOCATION_${upper}")
    endforeach()
    set(found_location FALSE)
    foreach(property IN LISTS properties)
        get_target_property(location Qt6::${target} "${property}")
        if(location)
            require_provider_path("Qt6::${target}/${property}" "${location}")
            set(found_location TRUE)
        endif()
    endforeach()
    if(NOT found_location)
        message(FATAL_ERROR "No imported executable/library for Qt6::${target}")
    endif()
endforeach()

if(APPLE)
    # Reuse the existing consumer-link adjustment; never patch provider binaries.
    include("${CMAKE_CURRENT_LIST_DIR}/../spike/cmake/LegacyQtMacSdk.cmake")
    set(probe_sdk "${CMAKE_OSX_SYSROOT}")
    if(NOT IS_ABSOLUTE "${probe_sdk}")
        if(NOT probe_sdk)
            set(probe_sdk macosx)
        endif()
        execute_process(COMMAND xcrun --sdk "${probe_sdk}" --show-sdk-path
            OUTPUT_VARIABLE probe_sdk OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    endif()
    set(probe_has_agl false)
    foreach(stub AGL AGL.tbd Versions/A/AGL Versions/A/AGL.tbd)
        if(EXISTS "${probe_sdk}/System/Library/Frameworks/AGL.framework/${stub}")
            set(probe_has_agl true)
        endif()
    endforeach()
    lc_spike_legacy_qt_agl("${Qt6_VERSION}" "${probe_has_agl}" PROBE_AGL_COMPAT)
endif()

add_executable(lcrhiproviderprobe providerprobe.cpp)
target_compile_features(lcrhiproviderprobe PRIVATE cxx_std_17)
target_link_libraries(lcrhiproviderprobe PRIVATE Qt6::Core Qt6::GuiPrivate Qt6::Widgets
    Qt6::Svg Qt6::SvgWidgets Qt6::UiTools Qt6::Test Qt6::OpenGL Qt6::OpenGLWidgets)
